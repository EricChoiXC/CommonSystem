## 1. 迁移脚本

`backend/src/main/resources/db/migration/V1_01_046__sys_datasource_schema.sql`（序号接续当前最新 `V1_01_045`）：

```sql
-- =====================================================================
-- V1_01_046: 数据源建表（可复用取数配置：查询能力 + 固定条件 + 返回字段）
-- 依据: docs/v1_8-数据源/数据源功能设计方案.md「§4.2 表结构」。
-- 模型: sys_datasource 自定义数据源单表（默认数据源由代码 Provider 注册，不入库）；
--       field_conditions / field_fields 对齐 sys_number_rule.field_rule 先例：
--       存储态为原文 JSON，load/save 请求响应为解析对象（禁止手工拼串）。
-- 约束: field_target 保存后锁定（改目标会使条件/字段白名单悬空，D3 先例）；
--       field_code 全局唯一（须与默认数据源编码共用命名空间，由应用层查重保证）；
--       关联（field_target → Provider）由应用层保证，不建物理外键（全工程惯例）。
-- =====================================================================

CREATE TABLE IF NOT EXISTS sys_datasource (
    field_id         VARCHAR(32)  NOT NULL COMMENT '主键ID',
    field_code       VARCHAR(100) NOT NULL COMMENT '数据源编码，全局唯一（与默认数据源共用命名空间），Service 层查重',
    field_name       VARCHAR(200) NOT NULL COMMENT '数据源名称',
    field_target     VARCHAR(100) NOT NULL COMMENT '目标查询能力编码（Provider code，如 calendar.list），保存后锁定',
    field_conditions JSON         DEFAULT NULL COMMENT '固定条件 JSON：[{key,op,value}]，value 支持 ${CURRENT_USER_ID} 等运行时变量',
    field_fields     JSON         DEFAULT NULL COMMENT '返回字段白名单 JSON：字符串数组，NULL=目标能力默认全字段',
    field_remark     VARCHAR(500) DEFAULT NULL COMMENT '备注',
    field_creator_id VARCHAR(32)  DEFAULT NULL COMMENT '创建者ID',
    field_create_time TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    field_updater_id VARCHAR(32)  DEFAULT NULL COMMENT '修改者ID',
    field_update_time TIMESTAMP   NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
    field_enable     BIT(1)       NOT NULL DEFAULT b'1' COMMENT '是否启用',
    PRIMARY KEY (field_id),
    UNIQUE KEY uk_datasource_code (field_code),
    KEY idx_datasource_target (field_target)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='数据源表（自定义数据源配置）';
```

```java
import java.util.Arrays;

/**
 * 数据源条件操作符
 *
 * @author Eric
 * @date 2026/11/13
 */
public enum SysDatasourceOp {

    /** 等于 */
    EQ,
    /** 不等于 */
    NE,
    /** 模糊匹配（前后 %） */
    LIKE,
    /** 大于 */
    GT,
    /** 大于等于 */
    GE,
    /** 小于 */
    LT,
    /** 小于等于 */
    LE,
    /** 在列表内 */
    IN;

    /**
     * 解析操作符文本，不区分大小写；非法返回 null（调用方报 CS_DATASOURCE_CONDITION_INVALID）
     */
    public static SysDatasourceOp of(String value) {
        if (null == value || value.isEmpty()) {
            return null;
        }
        return Arrays.stream(values())
                .filter(op -> op.name().equalsIgnoreCase(value))
                .findFirst().orElse(null);
    }
}
```

```java
import com.eric.base.common.utils.UserUtil;
import com.eric.base.sys.org.model.SysOrgUser;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.function.Supplier;

/**
 * 数据源固定条件运行时变量
 *
 * @author Eric
 * @date 2026/11/13
 * @description 固定条件 value 支持 ${xxx} 占位符，执行期由 SysDatasourceService 解析；
 *              变量随执行上下文求值（如当前登录人），保证「我创建的」按执行者动态翻译。
 */
public enum SysDatasourceVar {

    /** 当前登录人 ID（「我创建的」条件载体） */
    CURRENT_USER_ID(() -> {
        SysOrgUser user = UserUtil.getCurrentUser();
        return null == user ? null : user.getFieldId();
    }),
    /** 当天日期 yyyy-MM-dd */
    CURRENT_DATE(() -> LocalDate.now().format(DateTimeFormatter.ISO_LOCAL_DATE)),
    /** 当前时间 yyyy-MM-dd HH:mm:ss */
    CURRENT_DATETIME(() -> LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss")));

    private final Supplier<String> resolver;

    SysDatasourceVar(Supplier<String> resolver) {
        this.resolver = resolver;
    }

    /** 变量占位符形态：${CURRENT_USER_ID} */
    public String placeholder() {
        return "${" + name() + "}";
    }

    /** 求值；未登录时 CURRENT_USER_ID 求值为 null，由调用方走 UNAUTHORIZED */
    public String resolve() {
        return resolver.get();
    }

    /**
     * 判断文本是否本变量占位符
     */
    public boolean matches(String text) {
        return null != text && placeholder().equals(text.trim());
    }
}
```

```java
package com.eric.base.sys.datasource.spi;

import com.eric.base.common.base.resolver.BaseResolver;

import java.util.List;

/**
 * 数据源目标能力 SPI
 *
 * @author Eric
 * @date 2026/11/13
 * @description 业务模块实现本接口并注册为 Spring Bean，即向数据源模块声明一个
 *              「目标查询能力」；同时可在 {@link #getDefaultDatasources()} 中声明随模块
 *              提供的默认数据源。数据源模块经 {@code SysDatasourceRegistry} 收集全部实现：
 *              <ul>
 *              <li>targets/options 下拉的元数据来源（编码、名称、条件/返回字段定义）；</li>
 *              <li>自定义数据源的挂接白名单（field_target 只能取 getTargetCode）；</li>
 *              <li>统一执行入口的执行与鉴权委托（getTargetAction + getTargetResolver）。</li>
 *              </ul>
 */
public interface SysDatasourceProvider {

    /** 目标查询能力编码（全局唯一，命名 {模块}.{能力}，如 calendar.list） */
    String getTargetCode();

    /** 目标查询能力名称（下拉展示） */
    String getTargetName();

    /** 目标接口标识（ACTION_* 常量值，如 sys.calendar.list；执行期鉴权委托用） */
    String getTargetAction();

    /** 目标接口路径（如 /backend/sys/calendar/list；仅展示用） */
    String getTargetPath();

    /** 目标接口鉴权器（执行期经 AuthorizationHelper 委托目标 action 鉴权） */
    BaseResolver getTargetResolver();

    /** 条件字段定义（自定义数据源配置表单与执行期条件白名单） */
    List<SysDatasourceFieldDo> getConditionFields();

    /** 返回字段定义（默认全字段母集；投影白名单必须是其子集） */
    List<SysDatasourceFieldDo> getResultFields();

    /**
     * 执行查询
     *
     * @param context 条件合并后的执行上下文
     * @return 行列表（元素为 VO/模型对象，由数据源模块统一投影；总数经行级 count 一并返回）
     */
    SysDatasourceQueryResult query(SysDatasourceQueryContext context);

    /**
     * 随本能力提供的默认数据源（可多个）；返回空列表 = 仅作为目标能力，不带默认数据源
     *
     * @return 默认数据源定义列表
     */
    List<SysDatasourceDefaultDo> getDefaultDatasources();
}
```

### 2.3 Registry：默认数据源注册表

`sys/datasource/service/SysDatasourceRegistry.java`：

```java
package com.eric.base.sys.datasource.service;

import com.eric.base.sys.datasource.enums.SysDatasourceSourceType;
import com.eric.base.sys.datasource.spi.SysDatasourceDefaultDo;
import com.eric.base.sys.datasource.spi.SysDatasourceProvider;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * 数据源注册表
 *
 * @author Eric
 * @date 2026/11/13
 * @description 收集容器内全部 {@link SysDatasourceProvider}（方案 A3：默认数据源与目标能力
 *              由代码注册，对比 numberRule.yml 纯元数据先例——数据源需要可执行逻辑与目标
 *              鉴权器引用，只能由代码承载），构建三个进程内只读索引：
 *              <ul>
 *              <li>targetCode → Provider：自定义数据源挂接白名单与执行分发；</li>
 *              <li>defaultCode → {Provider, 默认定义}：默认数据源解析；</li>
 *              <li>编码占用全集：自定义数据源 field_code 查重的排除集（防伪装默认源）。</li>
 *              </ul>
 *              经 Spring 构造注入收集，无需追加 SysStartupTask 启动任务。
 */
@Service
public class SysDatasourceRegistry {

    private static final Logger log = LoggerFactory.getLogger(SysDatasourceRegistry.class);

    /** 能力编码 → Provider（LinkedHashMap 保序，供 targets 下拉稳定展示） */
    private final Map<String, SysDatasourceProvider> providersByTarget;

    /** 默认数据源编码 → 定义 */
    private final Map<String, SysDatasourceDefaultDo> defaultsByCode;

    /** 默认数据源编码 → 所属 Provider */
    private final Map<String, SysDatasourceProvider> providersByDefaultCode;

    public SysDatasourceRegistry(List<SysDatasourceProvider> providerList) {
        Map<String, SysDatasourceProvider> targets = new LinkedHashMap<>();
        Map<String, SysDatasourceDefaultDo> defaults = new LinkedHashMap<>();
        Map<String, SysDatasourceProvider> defaultOwners = new LinkedHashMap<>();
        for (SysDatasourceProvider provider : providerList) {
            if (null != targets.putIfAbsent(provider.getTargetCode(), provider)) {
                log.warn("duplicate datasource target code [{}], first provider wins", provider.getTargetCode());
                continue;
            }
            for (SysDatasourceDefaultDo def : provider.getDefaultDatasources()) {
                if (null != defaults.putIfAbsent(def.getCode(), def)) {
                    log.warn("duplicate default datasource code [{}], first definition wins", def.getCode());
                    continue;
                }
                defaultOwners.put(def.getCode(), provider);
            }
        }
        this.providersByTarget = Map.copyOf(targets);
        this.defaultsByCode = Map.copyOf(defaults);
        this.providersByDefaultCode = Map.copyOf(defaultOwners);
        log.info("datasource registry built: {} targets, {} default datasources",
                targets.size(), defaults.size());
    }

    /** 按能力编码取 Provider；不存在返回 null */
    public SysDatasourceProvider getProviderByTarget(String targetCode) {
        return null == targetCode ? null : providersByTarget.get(targetCode);
    }

    /** 按默认数据源编码取定义；不存在返回 null */
    public SysDatasourceDefaultDo getDefault(String code) {
        return null == code ? null : defaultsByCode.get(code);
    }

    /** 按默认数据源编码取所属 Provider；不存在返回 null */
    public SysDatasourceProvider getProviderByDefaultCode(String code) {
        return null == code ? null : providersByDefaultCode.get(code);
    }

    /**
     * 编码是否已被默认数据源占用（自定义数据源保存查重的排除集，防伪装默认源）
     */
    public boolean isCodeOccupiedByDefault(String code) {
        return null != code && defaultsByCode.containsKey(code);
    }

    /**
     * 全量目标能力 Provider 列表（targets 下拉）
     */
    public List<SysDatasourceProvider> listProviders() {
        return List.copyOf(providersByTarget.values());
    }

    /**
     * 全量默认数据源定义（options 下拉）；sourceType 固定 CODE
     */
    public List<SysDatasourceDefaultDo> listDefaults() {
        return List.copyOf(defaultsByCode.values());
    }

    /** 来源类型标记（供 options 区分 CODE/CUSTOM） */
    public static String typeOf(boolean custom) {
        return custom ? SysDatasourceSourceType.CUSTOM.getValue() : SysDatasourceSourceType.CODE.getValue();
    }

    /** 供日志/调试列出全部默认数据源编码 */
    public List<String> listDefaultCodes() {
        return new ArrayList<>(defaultsByCode.keySet());
    }
}
```

```java
package com.eric.base.sys.datasource.service;

import com.eric.base.common.base.model.Pages;
import com.eric.base.common.base.service.BaseService;
import com.eric.base.common.exception.AuthorizationException;
import com.eric.base.common.resolver.AuthorizationContext;
import com.eric.base.common.resolver.AuthorizationHelper;
import com.eric.base.common.utils.FieldProjector;
import com.eric.base.common.utils.StringUtil;
import com.eric.base.common.utils.UserUtil;
import com.eric.base.sys.datasource.dao.SysDatasourceDao;
import com.eric.base.sys.datasource.enums.SysDatasourceOp;
import com.eric.base.sys.datasource.enums.SysDatasourceVar;
import com.eric.base.sys.datasource.model.SysDatasource;
import com.eric.base.sys.datasource.spi.SysDatasourceConditionDo;
import com.eric.base.sys.datasource.spi.SysDatasourceDefaultDo;
import com.eric.base.sys.datasource.spi.SysDatasourceFieldDo;
import com.eric.base.sys.datasource.spi.SysDatasourceProvider;
import com.eric.base.sys.datasource.spi.SysDatasourceQueryContext;
import com.eric.base.sys.datasource.spi.SysDatasourceQueryResult;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.Date;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * 数据源业务服务
 *
 * @author Eric
 * @date 2026/11/13
 * @description 承载自定义数据源管理与统一执行编排：
 *              <ul>
 *              <li>管理：list/count/load/save/update/enable/delete，编码全局查重
 *                  （含默认数据源占用集），fieldTarget 保存后锁定，target 存在性校验；</li>
 *              <li>执行（方案 §3.1）：解析数据源（默认/自定义）→ 委托目标 action 鉴权 →
 *                  条件合并（固定覆盖调用方、白名单与字面量限制）→ Provider 查询 →
 *                  FieldProjector 投影 → 行列表 + metadata.total/fields。</li>
 *              </ul>
 */
@Service
public class SysDatasourceService extends BaseService<SysDatasource, SysDatasourceDao> {

    private final SysDatasourceRegistry registry;
    private final AuthorizationHelper authorizationHelper;

    public SysDatasourceService(SysDatasourceDao baseDao, SysDatasourceRegistry registry,
                                AuthorizationHelper authorizationHelper) {
        super(baseDao);
        this.registry = registry;
        this.authorizationHelper = authorizationHelper;
    }

    /* ==================== 统一执行 ==================== */

    /**
     * 统一执行入口：按 code 取数（方案 §3.2 时序）
     *
     * @param code       数据源编码（默认或自定义）
     * @param body       调用方过滤条件（固定条件同名覆盖之）
     * @param pages      分页参数
     * @param sortField  排序字段（须命中目标能力可排序白名单）
     * @param sortOrder  排序方向
     * @return 投影后的行列表与总数
     */
    public SysDatasourceExecution execute(String code, Map<String, Object> body, Pages pages,
                                          String sortField, String sortOrder) {
        // ① 解析数据源：默认 → Registry；自定义 → 表（须启用）
        Resolved resolved = resolve(code);

        // ② 委托目标 action 鉴权（A6）：如 sys.calendar.list → CalendarResolver
        authorizationHelper.authorize(resolved.provider.getTargetResolver(),
                new AuthorizationContext(resolved.provider.getTargetAction()));

        // ③ 条件合并：固定条件（解析变量）覆盖调用方条件，白名单与字面量限制校验（A7）
        Map<String, Object> conditions = mergeConditions(resolved, body);

        // ④ 排序字段白名单校验，非法回退 null（由目标能力按默认排序执行）
        String safeSortField = sanitizeSortField(resolved.provider, sortField);

        // ⑤ 执行查询
        SysDatasourceQueryContext context = new SysDatasourceQueryContext();
        context.setConditions(conditions);
        context.setPages(pages);
        context.setSortField(safeSortField);
        context.setSortOrder(sortOrder);
        SysDatasourceQueryResult result = resolved.provider.query(context);

        // ⑥ 字段投影（A4）：配置白名单 ∩ 目标能力返回字段母集
        List<String> fields = resolveOutputFields(resolved);
        List<Map<String, Object>> rows = new ArrayList<>();
        for (Object row : result.getRows()) {
            rows.add(FieldProjector.project(row, fields));
        }
        return new SysDatasourceExecution(rows, result.getTotal(), fields);
    }

    /**
     * 执行结果
     */
    public record SysDatasourceExecution(List<Map<String, Object>> rows, long total, List<String> fields) {
    }

    /**
     * 解析数据源：默认（Registry）优先，其次自定义（sys_datasource，须启用）；
     * 自定义 target 失效（Provider 下线残留）报 CS_DATASOURCE_TARGET_INVALID。
     */
    private Resolved resolve(String code) {
        SysDatasourceDefaultDo def = registry.getDefault(code);
        if (null != def) {
            SysDatasourceProvider provider = registry.getProviderByDefaultCode(code);
            return new Resolved(provider, def.getConditions(), def.getFields());
        }
        SysDatasource custom = getBaseDao().selectByCode(code);
        if (null == custom || !Boolean.TRUE.equals(custom.getFieldEnable())) {
            throw new IllegalArgumentException("CS_DATASOURCE_NOT_FOUND");
        }
        SysDatasourceProvider provider = registry.getProviderByTarget(custom.getFieldTarget());
        if (null == provider) {
            throw new IllegalArgumentException("CS_DATASOURCE_TARGET_INVALID");
        }
        return new Resolved(provider, parseConditions(custom.getFieldConditions()),
                parseFields(custom.getFieldFields()));
    }

    /**
     * 条件合并（方案 §6.3）：key ∈ 条件白名单、op 合法、literalAllowed=false 仅收运行时变量；
     * 固定条件优先，调用方同名条件丢弃；再补入调用方其余合法条件。
     */
    private Map<String, Object> mergeConditions(Resolved resolved, Map<String, Object> body) {
        Map<String, SysDatasourceFieldDo> whitelist = new LinkedHashMap<>();
        for (SysDatasourceFieldDo field : resolved.provider.getConditionFields()) {
            whitelist.put(field.getKey(), field);
        }
        Map<String, Object> merged = new LinkedHashMap<>();
        // 固定条件
        for (SysDatasourceConditionDo fixed : resolved.fixedConditions) {
            SysDatasourceFieldDo def = whitelist.get(fixed.getKey());
            if (null == def) {
                throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
            }
            merged.put(fixed.getKey(), resolveConditionValue(def, fixed.getOp(), fixed.getValue(), true));
        }
        // 调用方条件（固定条件已占用的 key 丢弃）
        if (null != body) {
            for (Map.Entry<String, Object> entry : body.entrySet()) {
                if (merged.containsKey(entry.getKey())) {
                    continue;
                }
                SysDatasourceFieldDo def = whitelist.get(entry.getKey());
                if (null == def || null == entry.getValue()) {
                    throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
                }
                merged.put(entry.getKey(), def.isLiteralAllowed()
                        ? entry.getValue()
                        : resolveConditionValue(def, "EQ", String.valueOf(entry.getValue()), true));
            }
        }
        return merged;
    }

    /**
     * 条件值解析：运行时变量求值（CURRENT_USER_ID 未登录由 AuthorizationHelper 前置拦截），
     * 字面量受 literalAllowed 限制；op 合法性一并校验。
     */
    private Object resolveConditionValue(SysDatasourceFieldDo def, String opText, String value, boolean fromConfig) {
        SysDatasourceOp op = SysDatasourceOp.of(opText);
        if (null == op || null == def.getOps() || !def.getOps().contains(op.name())) {
            throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
        }
        for (SysDatasourceVar var : SysDatasourceVar.values()) {
            if (var.matches(value)) {
                String resolved = var.resolve();
                if (null == resolved) {
                    throw new AuthorizationException("UNAUTHORIZED", "登录已过期，请重新登录");
                }
                return def.convert(resolved);
            }
        }
        if (!def.isLiteralAllowed()) {
            // 如 fieldCreatorId 只接受 ${CURRENT_USER_ID}：字面量一律拒绝，防越权（A7）
            throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
        }
        return def.convert(value);
    }

    /**
     * 有效输出字段：配置白名单（或默认数据源 fields）为空 = 目标能力全字段；
     * 非空时逐一校验 ∈ 返回字段母集，否则报 CS_DATASOURCE_FIELD_INVALID。
     */
    private List<String> resolveOutputFields(Resolved resolved) {
        Map<String, SysDatasourceFieldDo> resultFields = new LinkedHashMap<>();
        for (SysDatasourceFieldDo field : resolved.provider.getResultFields()) {
            resultFields.put(field.getKey(), field);
        }
        if (null == resolved.outputFields || resolved.outputFields.isEmpty()) {
            return new ArrayList<>(resultFields.keySet());
        }
        Set<String> keys = new LinkedHashSet<>();
        for (String key : resolved.outputFields) {
            if (!resultFields.containsKey(key)) {
                throw new IllegalArgumentException("CS_DATASOURCE_FIELD_INVALID");
            }
            keys.add(key);
        }
        return new ArrayList<>(keys);
    }

    /**
     * 排序字段白名单：取条件/返回字段定义中 sortable=true 的集合，不在集合内返回 null
     */
    private String sanitizeSortField(SysDatasourceProvider provider, String sortField) {
        if (StringUtil.isEmpty(sortField)) {
            return null;
        }
        for (SysDatasourceFieldDo field : provider.getResultFields()) {
            if (field.isSortable() && field.getKey().equals(sortField)) {
                return sortField;
            }
        }
        return null;
    }

    /* ==================== 管理（节选关键校验） ==================== */

    /**
     * 新建自定义数据源：编码全局查重（含默认数据源占用集）+ target 存在性校验
     */
    @Transactional
    public SysDatasource create(SysDatasource body) {
        validateForSave(body);
        SysOrgUser current = requireCurrentUser();
        body.initFieldId();
        body.setFieldCreatorId(current.getFieldId());
        body.setFieldCreateTime(new Date());
        getBaseDao().insert(body);
        return body;
    }

    /**
     * 编辑自定义数据源：fieldTarget 保存后锁定（以库值为准，不受理变更）
     */
    @Transactional
    public SysDatasource update(SysDatasource body) {
        if (StringUtil.isEmpty(body.getFieldId())) {
            throw new IllegalArgumentException("id 不能为空");
        }
        SysDatasource exist = getBaseDao().selectById(body.getFieldId());
        if (null == exist) {
            throw new IllegalArgumentException("数据源不存在");
        }
        validateFields(body);
        if (!exist.getFieldCode().equals(body.getFieldCode())) {
            // 编码可改，但须查重（排除自身 + 默认数据源占用）
            SysDatasource duplicated = getBaseDao().selectByCode(body.getFieldCode());
            if (null != duplicated && !duplicated.getFieldId().equals(body.getFieldId())) {
                throw new IllegalArgumentException("CS_DATASOURCE_CODE_DUPLICATED");
            }
            if (registry.isCodeOccupiedByDefault(body.getFieldCode())) {
                throw new IllegalArgumentException("CS_DATASOURCE_CODE_DUPLICATED");
            }
        }
        // fieldTarget 以库值为准（锁定）
        body.setFieldTarget(exist.getFieldTarget());
        body.setFieldUpdaterId(requireCurrentUser().getFieldId());
        body.setFieldUpdateTime(new Date());
        getBaseDao().update(body);
        return body;
    }

    private void validateForSave(SysDatasource body) {
        validateFields(body);
        if (null != getBaseDao().selectByCode(body.getFieldCode())
                || registry.isCodeOccupiedByDefault(body.getFieldCode())) {
            throw new IllegalArgumentException("CS_DATASOURCE_CODE_DUPLICATED");
        }
    }

    private void validateFields(SysDatasource body) {
        if (StringUtil.isEmpty(body.getFieldCode()) || StringUtil.isEmpty(body.getFieldName())) {
            throw new IllegalArgumentException("编码与名称不能为空");
        }
        if (StringUtil.isEmpty(body.getFieldTarget())
                || null == registry.getProviderByTarget(body.getFieldTarget())) {
            throw new IllegalArgumentException("CS_DATASOURCE_TARGET_INVALID");
        }
        // 条件与返回字段在保存期按目标能力定义做静态校验（同 §6.3 白名单口径）
        SysDatasourceProvider provider = registry.getProviderByTarget(body.getFieldTarget());
        validateConditions(provider, parseConditions(body.getFieldConditions()));
        validateOutputFields(provider, parseFields(body.getFieldFields()));
    }

    private void validateConditions(SysDatasourceProvider provider, List<SysDatasourceConditionDo> conditions) {
        if (null == conditions) {
            return;
        }
        Map<String, SysDatasourceFieldDo> whitelist = new HashMap<>();
        for (SysDatasourceFieldDo field : provider.getConditionFields()) {
            whitelist.put(field.getKey(), field);
        }
        for (SysDatasourceConditionDo condition : conditions) {
            SysDatasourceFieldDo def = whitelist.get(condition.getKey());
            if (null == def || null == SysDatasourceOp.of(condition.getOp())
                    || !def.getOps().contains(SysDatasourceOp.of(condition.getOp()).name())) {
                throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
            }
            boolean isVar = null != condition.getValue() && isVarPlaceholder(condition.getValue());
            if (!def.isLiteralAllowed() && !isVar) {
                throw new IllegalArgumentException("CS_DATASOURCE_CONDITION_INVALID");
            }
        }
    }

    private void validateOutputFields(SysDatasourceProvider provider, List<String> fields) {
        if (null == fields) {
            return;
        }
        Set<String> allowed = new LinkedHashSet<>();
        for (SysDatasourceFieldDo field : provider.getResultFields()) {
            allowed.add(field.getKey());
        }
        if (!allowed.containsAll(fields)) {
            throw new IllegalArgumentException("CS_DATASOURCE_FIELD_INVALID");
        }
    }

    private SysOrgUser requireCurrentUser() {
        SysOrgUser user = UserUtil.getCurrentUser();
        if (null == user) {
            throw new AuthorizationException("UNAUTHORIZED", "登录已过期，请重新登录");
        }
        return user;
    }

    /**
     * 判断文本是否任一运行时变量占位符（如 ${CURRENT_USER_ID}）
     */
    private static boolean isVarPlaceholder(String value) {
        for (SysDatasourceVar var : SysDatasourceVar.values()) {
            if (var.matches(value)) {
                return true;
            }
        }
        return false;
    }

    /**
     * 解析结果
     */
    private record Resolved(SysDatasourceProvider provider,
                            List<SysDatasourceConditionDo> fixedConditions,
                            List<String> outputFields) {
    }
}
```

> 说明：`parseConditions` / `parseFields` 为 Jackson 反序列化辅助（`Object` ↔ `List<SysDatasourceConditionDo>` / `List<String>`，损坏 JSON 按配置无效抛 `CS_DATASOURCE_CONDITION_INVALID`，先例 `SysNumberRuleService#load` 对 `field_rule` 的解析容错）；`SysDatasourceFieldDo#convert(String)` 为按 `type` 的类型转换（STRING 原样、NUMBER 转 BigDecimal、DATE/DATETIME 转 Date）。以上均属实现细节，落地时补齐单测。

```java
            case ACTION_OPTIONS, ACTION_TARGETS, ACTION_QUERY -> resolveLoggedIn();
```

### 4.2 消费示例：页面按数据源取数

任意列表页可改为「数据源驱动」：把数据源编码作为页面配置，取数统一走 `queryDatasource`，列头按 `metadata.fields` 渲染。以「我的日程」页取数片段为例（三段式列表页的列表区）：

```vue
<script setup>
import { ref, onMounted } from 'vue'
import { queryDatasource, getDatasourceOptions } from '@/api/datasource'
import { ElMessage } from 'element-plus'

// 页面绑定的数据源编码（可来自路由 meta 或页面配置）
const DATASOURCE_CODE = 'calendar.my-created'

const loading = ref(false)
const rows = ref([])
const columns = ref([]) // [{ key, label }]，来自数据源字段定义
const total = ref(0)
const pages = ref({ pageNo: 1, pageSize: 20, needPage: true })
const filters = ref({ fieldName: '' })
const sortField = ref('startTime')
const sortOrder = ref('desc')

/** 查询：固定条件（我创建的）由服务端强制，前端只传增量过滤条件 */
async function query() {
  loading.value = true
  try {
    const res = await queryDatasource({
      code: DATASOURCE_CODE,
      body: { ...filters.value },
      pages: pages.value,
      sortField: sortField.value,
      sortOrder: sortOrder.value
    })
    rows.value = res.data || []
    total.value = Number(res.metadata?.total ?? 0)
  } catch (e) {
    ElMessage.error(e.message || '查询失败')
  } finally {
    loading.value = false
  }
}

/** 按数据源字段定义渲染列头（options 返回有效字段集，与执行结果 fields 一致） */
onMounted(async () => {
  try {
    const res = await getDatasourceOptions()
    const option = (res.data || []).find((item) => item.code === DATASOURCE_CODE)
    columns.value = (option?.fields || []).map((f) => ({ key: f.key, label: f.label }))
  } catch (e) {
    columns.value = []
  }
  await query()
})
</script>

<template>
  <el-table v-loading="loading" :data="rows">
    <el-table-column
      v-for="col in columns"
      :key="col.key"
      :prop="col.key"
      :label="col.label"
    />
  </el-table>
  <el-pagination
    v-model:current-page="pages.pageNo"
    v-model:page-size="pages.pageSize"
    layout="total, prev, pager, next, sizes"
    :page-sizes="[10, 20, 50, 100]"
    :total="total"
    @current-change="query"
    @size-change="pages.pageNo = 1; query()"
  />
</template>
```

> 也可封装 `components/DatasourceTable.vue` 通用组件（props：`code`、可选 `filters` / `sortField`），内部完成 options → 列头 → query 的编排，供更多页面复用；组件契约须先补 `docs/系统模块/前端公共组件.md`。

### 4.3 管理页要点

`views/datasource/DatasourceManageView.vue` + `DatasourceFormView.vue` 按规范实现，关键差异点：

1. **列表页**：三段式（筛选区 `fieldName`/`fieldCode`/`fieldTarget`/`fieldEnable` → 工具区（左排序仅创建时间，右查询/新建）→ 列表区）。列：编码、名称、目标能力、固定条件数、返回字段数、启用、创建时间、操作（编辑/启停/删除，删除二次确认）。不展示主键，操作列 `fixed="right"`。
2. **表单页**：
   - 目标能力：`el-select`，选项来自 `getDatasourceTargets()`（`code（name）` 展示）；**编辑态只读锁定**（对齐编号规则「所属接入项」D3 先例）；
   - 固定条件：按所选能力的 `conditionFields` 动态渲染行（字段下拉 + 操作符下拉 + 值输入）；`literalAllowed=false` 的字段（如「创建人」）值固定为 `${CURRENT_USER_ID}`，展示为只读 tag「当前登录人」，不给输入框；
   - 返回字段：`el-select multiple`，选项来自 `resultFields`，空选 = 默认全字段（表单内提示）；
   - 新建 `onMounted` 调 `api/common.js` 的 `getInitID()` 取 32 位 ID 赋 `fieldId`；编辑按路由 id 调 `loadDatasource` 回填。
3. **路由与菜单**：`/datasource`、`/datasource/create`、`/datasource/edit/:id`、`/datasource/detail/:id`（动态路由 `meta.tab: 'name'`）；`HomeView.vue` 补菜单项与 `activeMenu` 前缀归约；菜单显示权限挂 `SYS_DATASOURCE_ADMIN`。
